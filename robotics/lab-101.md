PWM 
-----------------------------------
```
from machine import Pin, PWM
from time import sleep_ms

led = PWM(Pin(2), freq=1000, duty_u16=0)
STEPS = 256
# freq=1000 sets the PWM frequency to 1000 Hz.
# The pin switches fully on and off 1000 times per second.
# Gamma is the exponent in a power curve.
# It makes up for the fact that your eyes don't see brightness linearly.
GAMMA = 2.2

def level(i):
    return int(((i / (STEPS - 1)) ** GAMMA) * 65535)

while True:
    for i in range(STEPS):
        led.duty_u16(level(i))
        sleep_ms(5)
    for i in range(STEPS - 1, -1, -1):
        led.duty_u16(level(i))
        sleep_ms(5)



`````


Temperature and Humidity
--------------------------

```
from machine import Pin   # lets our code use the ESP32's pins
import dht                # a helper that knows how to talk to DHT sensors
import time               # lets us pause the program

sensor = dht.DHT11(Pin(2))   # our DHT11 sensor is wired to pin 4

while True:               # do this forever
    try:                  # "try" means: attempt this, but don't crash if it fails
        sensor.measure()          # tell the sensor: take a fresh reading now
        temp_c = sensor.temperature()   # get the temperature in Celsius
        hum = sensor.humidity()         # get the humidity as a percent
        temp_f = temp_c * 9 / 5 + 32    # math to change Celsius into Fahrenheit

        # print a neat line. {:.1f} means show 1 number after the dot (like 23.4)
        print("Temp: {:.1f} C / {:.1f} F   Humidity: {:.1f} %".format(
            temp_c, temp_f, hum
        ))
    except OSError:       # if the sensor didn't answer (loose wire, too fast)...
        print("Sensor read failed, retrying...")   # ...just say so and keep going

    time.sleep(2)         # wait 2 seconds. The DHT11 is slow and needs a rest

```

```
from machine import Pin
from time import sleep

led = Pin(2, Pin.OUT)


def blink(duration):
    led.on()
    sleep(duration)
    led.off()
    sleep(0.2)


while True:
    for i in range(3):
        blink(0.2)
    sleep(0.4)

    for i in range(3):
        blink(0.6)
    sleep(0.4)

    for i in range(3):
        blink(0.2)
    sleep(2)
    ```
