Blinking

```
from machine import Pin
from time import sleep
led = Pin(2, Pin.OUT)

while True:
  led.value(1)
  sleep(0.5)
  led.value(0)
  sleep(0.5)
```



SOS

```
from machine import Pin
from time import sleep

led = Pin(2, Pin.OUT)


def blink(duration):
    led.on()
    sleep(duration)
    led.off()
    sleep(0.2)


while True:
    for i in range(3):
        blink(0.2)
    sleep(0.4)

    for i in range(3):
        blink(0.6)
    sleep(0.4)

    for i in range(3):
        blink(0.2)
    sleep(2)
    ```
