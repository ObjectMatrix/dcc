### Connect ESP32 to a Wi-Fi, send DH11 data to `https://thingspeak.mathworks.com/`

```
from machine import Pin                      # lets our code use the ESP32's pins
import dht                                   # a helper that knows how to talk to DHT sensors
import time                                  # lets us pause the program
import network                               # lets the ESP32 join WiFi
import urequests                             # lets the ESP32 send data to a website

WIFI_SSID = "WIFINAME"                       # <-- your WiFi name
WIFI_PASSWORD = "PASSWORD"                   # <-- your WiFi password
THINGSPEAK_KEY = "SPHJO5R26GMQQ552AZ"        # <--YOUR_THINGSPEAK_WRITE_KEY:  your secret upload key (keep private)
myname = "Robot-707"                         # < -- D6QS04Z0D1DFYCET YOUR_THINGSPEAK_READ_KEY

# wifi; Connect once
wlan = network.WLAN(network.STA_IF)          # turn the ESP32 into a WiFi device
if not wlan.isconnected():
    wlan.disconnect()
wlan.active(True)                            # switch the WiFi radio on
wlan.connect(WIFI_SSID, WIFI_PASSWORD)       # join your network
while not wlan.isconnected():                # keep waiting until we're connected
    time.sleep(1)
print ("Connection to WiFi is Successful")

# our DHT11 sensor is wired to pin 2
sensor = dht.DHT11(Pin(2))   

while True:                                    # do this forever
    try:                                       # "try" means: attempt this, but don't crash if it fails
        sensor.measure()                       # tell the sensor: take a fresh reading now
        temp_c = sensor.temperature()          # get the temperature in Celsius
        hum = sensor.humidity()                # get the humidity as a percent
        temp_f = temp_c * 9 / 5 + 32           # math to change Celsius into Fahrenheit
        

        # Inside loop after temp/hum are read
        # Build a web address that carries our two numbers, then visit it to save them online.
        url = "https://api.thingspeak.com/update?api_key={}&field1={}&field2={}".format(
            THINGSPEAK_KEY, temp_f, hum
        )
        r = urequests.get(url)                    # send the data up to the ThingSpeak website
        r.close() 
        # print a neat line. {:.1f} means show 1 number after the dot (like 23.4)
        print("{:s} | Temp: {:.1f} C / {:.1f} F   Humidity: {:.1f} %".format(
            myname, temp_c, temp_f, hum
        ))
    except OSError:                                # if the sensor didn't answer (loose wire, too fast)
        print("Sensor read failed, retrying...")   # ...just say so and keep going

    time.sleep(2)                                  # wait 2 seconds. The DHT11 is slow and needs a rest


```

# Connect ESP32 to Bluetooth and Test it with nRF connect

```
from machine import Pin
import bluetooth

# ---------------------------------
# 1. STUDENT SETTINGS
# ---------------------------------

# Each student changes this name
STUDENT_NAME = "ROBOT-01"

# LED connected to GPIO2
led = Pin(2, Pin.OUT)
led.off()

# ---------------------------------
# 2. START BLUETOOTH
# ---------------------------------

ble = bluetooth.BLE()
ble.active(True)

# Bluetooth service and characteristic
SERVICE_UUID = bluetooth.UUID(0xA000)
COMMAND_UUID = bluetooth.UUID(0xA001)

# Allow phone to write commands
services = (
    (
        SERVICE_UUID,
        (
            (
                COMMAND_UUID,
                bluetooth.FLAG_WRITE | bluetooth.FLAG_WRITE_NO_RESPONSE,
            ),
        ),
    ),
)

# Register Bluetooth service
((command_handle,),) = ble.gatts_register_services(services)

# ---------------------------------
# 3. ADVERTISE DEVICE NAME
# ---------------------------------

name = STUDENT_NAME.encode()

#  b"\x02\x01\x06" can also be written as bytes([2,1,6])

advertising_data = (
    b"\x02\x01\x06"
    + bytes([len(name) + 1, 0x09])
    + name
)


def start_advertising():
    ble.gap_advertise(
        100000,
        adv_data=advertising_data,
    )
    print("Bluetooth ready:", STUDENT_NAME)


# ---------------------------------
# 4. HANDLE PHONE COMMANDS
# ---------------------------------

def bluetooth_event(event, data):
    # Phone connected
    if event == 1:
        print("Phone connected!")

    # Phone disconnected
    elif event == 2:
        print("Phone disconnected")
        start_advertising()

    # Phone sent a command
    elif event == 3:
        conn_handle, attr_handle = data

        if attr_handle == command_handle:
            message = ble.gatts_read(command_handle)

            if message == b"1":
                led.on()
                print("LED ON")

            elif message == b"0":
                led.off()
                print("LED OFF")

            else:
                print("Unknown command:", message)


# Register Bluetooth event handler
ble.irq(bluetooth_event)

# ---------------------------------
# 5. START
# ---------------------------------

start_advertising()

print("Waiting for phone commands...")
```

## PWM (Part 1: Understand it, Build the circuit, Use D2, GND and a LED)
PWM is the foundation of robotics:  
- 🚗 Wheels and motors, 
- 🦾 Robot arms and servos,
- 🚁 Drones,
- 💡 Lights and eyes,
- Sounds.
  
PWM explained 💡   
Imagine a light switch. 
A normal light switch only does two things: ON or OFF. So how could you make a light half bright using just that switch?  

The trick: flick it really, really fast!  
Say you flick the switch on and off 1000 times every second. That's way faster than your eyes can follow. Your eyes can't see each flick, so they just see the average:  

🔆 Mostly ON → looks bright.  
🌗 Half ON, half OFF → looks medium.  
🌑 Mostly OFF → looks dim.  

That's PWM.   
It stands for Pulse Width Modulation, a fancy way of saying "change how long each ON flick lasts."   

-----------------------------------
```
from machine import Pin, PWM
from time import sleep_ms

led = PWM(Pin(2), freq=1000, duty_u16=0)
STEPS = 256
GAMMA = 2.2

# duty_u16=0 sets the starting duty cycle when the PWM is created.
# 0 means the pin stays OFF the whole time, so the LED starts dark.
# freq=1000 sets the PWM frequency to 1000 Hz.
# The pin switches fully on and off 1000 times per second.
# Gamma is the exponent in a power curve.
# It makes up for the fact that your eyes don't see brightness linearly.



def level(i):
    return int(((i / (STEPS - 1)) ** GAMMA) * 65535)

while True:
    for i in range(STEPS):
        led.duty_u16(level(i))
        sleep_ms(5)
    for i in range(STEPS - 1, -1, -1):
        led.duty_u16(level(i))
        sleep_ms(5)



`````
